# 角色和人设板块修复方案

## 方案概述

本方案基于《角色和人设板块深度缺陷分析报告》，提供系统性的修复路径。采用**渐进式重构**策略，确保在修复过程中系统保持稳定运行。

**核心目标：**
1. 统一三个并行系统，建立单一数据源
2. 修复高危安全和稳定性缺陷
3. 提升代码可维护性和可测试性
4. 完善用户体验

**实施周期：** 4-6周（分阶段交付）

### 1.1 统一数据模型 - affinity字段类型修复 🔴

**问题：** `affinity`在`CompoundEmotionalState`中是`int`（0-8级），在`CharacterState`中是`float`，两者不同步。

**修复方案：**

```python
# 1. 创建统一的数据模型定义文件：shisi/character/unified_models.py

from enum import IntEnum
from typing import Optional
from pydantic import BaseModel, Field, field_validator

class AffinityLevel(IntEnum):
    """统一的好感度等级定义"""
    STRANGER = 0      # 陌生人
    ACQUAINTANCE = 1  # 认识
    FRIEND = 2        # 朋友
    GOOD_FRIEND = 3   # 好朋友
    CONFIDANT = 4     # 知己
    AMBIGUOUS = 5     # 暧昧
    LOVER = 6         # 恋人
    PASSIONATE = 7    # 热恋
    BOND = 8          # 羁绊

class UnifiedEmotionalState(BaseModel):
    """统一的情感状态模型"""
    primary_emotion: str = "平常"
    primary_intensity: float = Field(default=0.5, ge=0.0, le=1.0)
    energy: float = Field(default=1.0, ge=0.0, le=1.0)
    affinity_level: AffinityLevel = AffinityLevel.STRANGER
    affection_points: float = Field(default=0.0, ge=0.0)
    
    @field_validator('affection_points')
    @classmethod
    def validate_affection_points(cls, v: float) -> float:
        return round(v, 1)  # 保留1位小数
    
    def to_character_state_dict(self) -> dict:
        """转换为CharacterState兼容格式"""
        return {
            "affinity": float(self.affinity_level.value),  # 统一转为float
            "emotion_stage": self.affinity_level.name,
            "affection_points": self.affection_points,
        }

class UnifiedCharacterState(BaseModel):
    """统一的角色状态模型"""
    character_id: str
    name: str
    format: str = "chara_card_v2"
    is_active: bool = False
    affinity_level: AffinityLevel = AffinityLevel.STRANGER
    affection_points: float = 0.0
    emotion_stage: str = "陌生人"
    avatar_url: Optional[str] = None
    tags: list[str] = Field(default_factory=list)

    @classmethod

        data = info.data
        if 'affinity_level' in data:
            return data['affinity_level'].name
        return v
```

```python
async def validate_upload_file(file: UploadFile) -> tuple[bytes, str]:
    """
    验证上传文件的安全性
    
    Returns:
        tuple: (文件内容, 检测到的MIME类型)
    
    Raises:
        FileValidationError: 验证失败
    """
    # 1. 检查文件名
    if not file.filename:
        raise FileValidationError("文件名不能为空")
    
    # 2. 检查文件扩展名
    file_ext = Path(file.filename).suffix.lower()
    if file_ext not in ALLOWED_EXTENSIONS:
        raise FileValidationError(f"不支持的文件类型: {file_ext}")
    
    # 3. 读取内容（限制大小）
    content = await file.read(MAX_FILE_SIZE + 1)
    if len(content) > MAX_FILE_SIZE:
        raise FileValidationError(f"文件大小超过限制: {MAX_FILE_SIZE / 1024 / 1024}MB")
    
    # 4. MIME类型检测
    mime = magic.Magic(mime=True)
    detected_mime = mime.from_buffer(content)
    
    # 对于JSON文件，允许text/plain（某些系统识别问题）
    if file_ext == '.json' and detected_mime == 'text/plain':
        # 进一步验证是否为有效JSON
        try:
            json.loads(content.decode('utf-8'))
            detected_mime = 'application/json'
        except (json.JSONDecodeError, UnicodeDecodeError):
            raise FileValidationError("文件内容不是有效的JSON")
    
    if detected_mime not in ALLOWED_MIME_TYPES and file_ext != '.png':
        raise FileValidationError(f"不安全的文件类型: {detected_mime}")
    
    # 5. 内容安全扫描（基础）
    if file_ext == '.json':
        content_str = content.decode('utf-8', errors='ignore')
        # 检查危险模式
        dangerous_patterns = [
            '__import__',
            'eval(',
            'exec(',
            'os.system',
            'subprocess',
        ]
        for pattern in dangerous_patterns:
            if pattern in content_str:
                raise FileValidationError(f"检测到潜在危险内容: {pattern}")
    
    return content, detected_mime
```

### 2.2 重构缓存机制

**修复方案：**

```python
# my_character/utils/cache.py - 新的缓存工具模块

import hashlib
import json
from typing import TypeVar, Generic, Optional
from dataclasses import dataclass
from datetime import datetime, timedelta
import threading

T = TypeVar('T')

@dataclass
class CacheEntry(Generic[T]):
    """缓存条目"""
    value: T
    created_at: datetime
    ttl_seconds: Optional[int] = None
    
    def is_expired(self) -> bool:
        if self.ttl_seconds is None:
            return False
        return datetime.now() > self.created_at + timedelta(seconds=self.ttl_seconds)

class ContentHashCache(Generic[T]):
    """
    基于内容哈希的缓存
    解决原方案中使用字符串长度作为缓存键的问题
    """
    
    def __init__(
        self, 
        max_size: int = 100,
        default_ttl_seconds: Optional[int] = None,
    ):
        self._max_size = max_size
        self._default_ttl = default_ttl_seconds
        self._cache: dict[str, CacheEntry[T]] = {}
        self._lock = threading.RLock()
        self._access_order: list[str] = []  # LRU顺序
    
    def _generate_key(self, **kwargs) -> str:
        """生成内容哈希键"""
        # 将参数序列化为JSON并计算哈希
        content = json.dumps(kwargs, sort_keys=True, default=str, ensure_ascii=False)
        return hashlib.sha256(content.encode('utf-8')).hexdigest()
    
    def get(self, **kwargs) -> Optional[T]:
        """获取缓存值"""
        key = self._generate_key(**kwargs)
        
        with self._lock:
            entry = self._cache.get(key)
            if entry is None:
                return None
            
            if entry.is_expired():
                del self._cache[key]
                self._access_order.remove(key)
                return None
            
            # 更新LRU顺序
            self._access_order.remove(key)
            self._access_order.append(key)
            
            return entry.value
    
    def set(self, value: T, ttl_seconds: Optional[int] = None, **kwargs) -> None:
        """设置缓存值"""
        key = self._generate_key(**kwargs)
        ttl = ttl_seconds or self._default_ttl
        
        with self._lock:
            # 清理过期条目
            self._cleanup_expired()
            
            # 如果键已存在，更新访问顺序
            if key in self._cache:
                self._access_order.remove(key)
            
            # 添加新条目
            self._cache[key] = CacheEntry(
                value=value,
                created_at=datetime.now(),
                ttl_seconds=ttl,
            )
            self._access_order.append(key)
            
            # LRU淘汰
            self._evict_if_needed()
    
    def _cleanup_expired(self) -> None:
        """清理过期条目"""
        expired_keys = [
            key for key, entry in self._cache.items() 
            if entry.is_expired()
        ]
        for key in expired_keys:
            del self._cache[key]
            self._access_order.remove(key)
    
    def _evict_if_needed(self) -> None:
        """LRU淘汰"""
        while len(self._cache) > self._max_size:
            oldest_key = self._access_order.pop(0)
            del self._cache[oldest_key]
    
    def invalidate(self, pattern: Optional[str] = None) -> int:
        """使缓存失效"""
        with self._lock:
            if pattern is None:
                count = len(self._cache)
                self._cache.clear()
                self._access_order.clear()
                return count
            
            # 根据模式删除
            keys_to_remove = [
                key for key in self._cache.keys() 
                if pattern in key
            ]
            for key in keys_to_remove:
                del self._cache[key]
                self._access_order.remove(key)
            
            return len(keys_to_remove)
    
    def stats(self) -> dict:
        """获取缓存统计"""
        with self._lock:
            return {
                "size": len(self._cache),
                "max_size": self._max_size,
                "hit_rate": getattr(self, '_hit_rate', 0),  # 需要额外实现统计
            }


# 在PersonaEngine中使用新缓存
class PersonaEngine:
    def __init__(self, ...):
        # ... 其他初始化 ...
        
        # 使用新的内容哈希缓存
        self._prompt_cache = ContentHashCache[str](
```

```python
import sqlite3
from pathlib import Path
import sys

def run_migrations(db_path: Path):
    """运行所有迁移脚本"""
    migrations_dir = Path("migrations")
    
    conn = sqlite3.connect(str(db_path))
    try:
        # 创建迁移记录表
        conn.execute("""
            CREATE TABLE IF NOT EXISTS schema_migrations (
                version TEXT PRIMARY KEY,
                applied_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
            )
        """)
        
        # 获取已应用的迁移
        applied = set(
            row[0] for row in 
            conn.execute("SELECT version FROM schema_migrations")
        )
        
        # 运行未应用的迁移
        for migration_file in sorted(migrations_dir.glob("*.sql")):
            version = migration_file.stem
            if version in applied:
                print(f"跳过已应用的迁移: {version}")
                continue
            
            print(f"应用迁移: {version}")
            with open(migration_file, 'r') as f:
                conn.executescript(f.read())
            
            conn.execute(
                "INSERT INTO schema_migrations (version) VALUES (?)",
                (version,)
            )
            conn.commit()
        
        print("所有迁移完成")
        
    except Exception as e:
        conn.rollback()
        print(f"迁移失败: {e}", file=sys.stderr)
        sys.exit(1)
    finally:
        conn.close()

if __name__ == "__main__":
    db_path = Path("data/sqlite.db")
    run_migrations(db_path)
```
