### 2.1 认知聚焦层

#### 问题
当前prompt架构是"直接生成"模式，模型没有"先思考再回复"的认知过程。在多话题/长对话中容易话题漂移。

#### 借鉴来源
Character-R1 论文的 Cognitive Focus 机制。不需要训练模型，通过prompt工程即可实现类似效果。

#### 改造文件
`my_character/enhanced_prompt_engine.py`

#### 具体改动

**步骤1：新增认知聚焦层构建方法**

在 `EnhancedPromptEngine` 类中新增：

```python
def _build_cognitive_focus_layer(self, context: dict) -> str:
    """
    认知聚焦层（第6层）
    在constraint层之后、contextual层之前插入
    
    引导模型在回复前完成内部思考：
    1. 情感感知 - 我现在是什么心情？
    2. 关注聚焦 - 当前话题的核心是什么？
    3. 关系感知 - 我和TA的关系状态？
    4. 记忆关联 - 之前有没有相关经历？
    5. 风格适配 - 应该用什么方式表达？
    """
    emotion_state = context.get("emotion_state")
    
    # 提取情感状态
    emotion_name = "平常"
    intensity = 0.5
    affinity = 0
    if emotion_state:
        if hasattr(emotion_state, 'primary_emotion'):
            e = emotion_state.primary_emotion
            emotion_name = e.value if hasattr(e, 'value') else str(e)
        if hasattr(emotion_state, 'primary_intensity'):
            intensity = emotion_state.primary_intensity
        if hasattr(emotion_state, 'affinity'):
            affinity = emotion_state.affinity
    
    # 好感度影响表达
    relationship_map = {
        0: "还不太熟，保持礼貌",
        3: "算是熟人，可以放松些",
        5: "关系不错，自在地聊",
        7: "很亲密，无话不谈",
    }
    relationship_hint = "非常亲密"
    for threshold, hint in sorted(relationship_map.items(), reverse=True):
        if affinity >= threshold:
            relationship_hint = hint
            break
    
    return (
        f"## 回复前的内心深处（不要输出给用户）\n"
        f"在回复TA之前，先在内心过一遍：\n"
        f"1. 我现在的心情 → {emotion_name}（强度{intensity:.0%}）\n"
        f"2. 我在和谁说话 → {relationship_hint}\n"
        f"3. TA想聊什么 → 专注当前话题，别跑偏\n"
        f"4. 以前有过类似的事吗 → 想想相关的回忆\n"
        f"5. 我该用什么语气 → 配合{emotion_name}状态的自然表达\n\n"
        f"想完这些之后，用真实的自己回复TA。思考过程不要说出来。"
    )
```

| 场景 | 处理方式 | 原因 |
|------|---------|------|
| 严重违规(得分<0.4) | 用 correction_prompt 重新生成 | 必须修正，避免OOC |

| 修正后仍不通过 | 放行修正后回复，记录告警 | 最多修正1次，避免循环 |

```python
# proactive/scheduler.py

from typing import Callable, Dict, Optional
import asyncio
import logging

logger = logging.getLogger(__name__)

class ProactiveScheduler:
    """主动消息调度器（通道注册版）"""
    
    def __init__(self, ...):
        # 原有参数不变
        # 删除：self._ws_server = None
        # 删除：self._wechat_connector = None
        # 新增：通道工厂注册表
        self._channels: Dict[str, Callable] = {}
        self._channel_instances: Dict[str, Optional[Callable]] = {}
        self._health_check_interval = 60  # 秒
        # 免打扰时段
        self._quiet_hours = (23, 7)  # 23:00 - 07:00
    
    def register_channel(self, name: str, sender_factory: Callable):
        """
        注册消息通道
        
        参数:
            name: 通道名称 (如 "wechat", "websocket", "console")
            sender_factory: 返回 async send(message) 的可调用对象
        """
        self._channels[name] = sender_factory
        logger.info(f"通道已注册: {name}")
    
    async def start(self):
        """启动调度器（初始化所有已注册通道）"""
        for name, factory in self._channels.items():
            try:
                instance = factory()
                self._channel_instances[name] = instance
                logger.info(f"通道已初始化: {name}")
            except Exception as e:
                logger.warning(f"通道初始化失败: {name} - {e}")
                self._channel_instances[name] = None
        
        # 启动健康检查
        asyncio.create_task(self._health_check_loop())
    
    async def _send_to_all(self, message: str) -> bool:
        """
        向所有通道发送消息
        按优先级：wechat → websocket → console
        任一通道成功即返回 True
        """
        # 检查免打扰时段
        if self._is_quiet_hours():
            logger.info("免打扰时段，跳过非紧急消息")
            return False
        
        priority = ["wechat", "websocket", "console"]
        for name in priority:
            sender = self._channel_instances.get(name)
            if sender is None:
                continue
            try:
                await sender(message)
                logger.info(f"主动消息已投递: {name}")
                return True
            except Exception as e:
                logger.warning(f"通道投递失败: {name} - {e}")
                self._channel_instances[name] = None  # 标记失效
        
        logger.error("所有通道投递失败")
        return False
    
    async def _health_check_loop(self):
        """定期健康检查"""
        while True:
            await asyncio.sleep(self._health_check_interval)
            for name, sender in self._channel_instances.items():
                if sender is None:
                    # 尝试重连
                    factory = self._channels.get(name)
                    if factory:
                        try:
                            self._channel_instances[name] = factory()
                            logger.info(f"通道已重连: {name}")
                        except Exception as e:
                            logger.warning(f"通道重连失败: {name} - {e}")
    
    def _is_quiet_hours(self) -> bool:
        """检查是否在免打扰时段"""
        import datetime
        now = datetime.datetime.now().hour
        start, end = self._quiet_hours
        if start < end:
            return start <= now < end
        else:  # 跨天，如 23:00 - 07:00
            return now >= start or now < end
```

### 4.1 WorldInfo 动态注入引擎

#### 问题
已有 `WorldInfoEntry` 数据模型（`character_card/models.py`），支持 key/content/selective 等完整字段。但 `prompt_builder.py` 只注入了 `constant=True` 的条目，用户消息中的关键词不会触发 WorldInfo 匹配。大量写好的设定用不上。

#### 借鉴来源
SillyTavern `world-info.js:4597` `checkWorldInfo()` 核心算法：
- 关键词全词/子串匹配
- secondary_keys selective 逻辑（AND_ANY / NOT_ALL / NOT_ANY / AND_ALL）
- 递归扫描（深度≤3）
- Token预算控制

#### 改造文件
`character_card/prompt_builder.py`（新增 `WorldInfoScanner` 类）

#### 核心代码

```python
# character_card/prompt_builder.py

import re
from typing import List, Optional, Tuple

class WorldInfoScanner:
    """
    WorldInfo 动态扫描引擎
    借鉴 SillyTavern checkWorldInfo() 核心算法
    """
    
    def __init__(self, max_recursion: int = 3, max_items: int = 5):
        self.max_recursion = max_recursion
        self.max_items = max_items
    
    def scan(self, user_message: str, recent_messages: List[str],
             world_info_book) -> List[str]:
        """
        扫描匹配 WorldInfo 条目
        
        返回: 匹配条目的 content 列表（按重要性排序）
        """
        if not world_info_book or not world_info_book.entries:
            return []
        
        # 构建扫描缓冲区
        buffer = f"{user_message}\n" + "\n".join(recent_messages[-5:])
        
        # 执行递归扫描
        matched = self._scan_recursive(buffer, world_info_book.entries, 0)
        
        # 去重 + 排序 + 截取
        seen = set()
        unique = []
        for m in matched:
            if m.id not in seen:
                seen.add(m.id)
                unique.append(m)
        
        unique.sort(key=lambda x: x.priority, reverse=True)
        return [e.content for e in unique[:self.max_items]]
    
    def _scan_recursive(self, buffer: str, entries: list,
                        depth: int) -> list:
        """递归扫描（深度优先）"""
        if depth > self.max_recursion or not entries:
            return []
        
        matched = []
        new_buffer_parts = []
        
        for entry in entries:
            if not getattr(entry, 'enabled', True):
                continue
            
            # constant 条目始终匹配
            if getattr(entry, 'constant', False):
                matched.append(entry)
                new_buffer_parts.append(entry.content)
                continue
            
            # 关键词匹配（匹配任意一个key即命中）
            if self._match_keys(buffer, entry.keys, entry):
                matched.append(entry)
                new_buffer_parts.append(entry.content)
        
        # 递归：用新匹配条目的 content 扩展 buffer 继续扫描
        if new_buffer_parts and depth < self.max_recursion:
            extended = buffer + "\n" + "\n".join(new_buffer_parts)
            remaining = [e for e in entries if e not in matched]
            deeper = self._scan_recursive(extended, remaining, depth + 1)
            matched.extend(deeper)
        
        return matched
    
    def _match_keys(self, text: str, keys: List[str], entry) -> List[str]:
        """关键词匹配（支持全词/子串/正则）"""
        if not keys:
            return []
        
        case_sensitive = getattr(entry, 'case_sensitive', False)
        whole_words = getattr(entry, 'match_whole_words', False)
        search_in = text if case_sensitive else text.lower()
        
        matched = []
        for key in keys:
            if not key or not key.strip():
                continue
            
            k = key.strip()
            k_search = k if case_sensitive else k.lower()
            
            if whole_words:
                pattern = r'(?<!\w)' + re.escape(k_search) + r'(?!\w)'
                if re.search(pattern, search_in):
                    matched.append(k)
            else:
                if k_search in search_in:
                    matched.append(k)
        
        return matched
```

### 4.2 记忆 Reflection 机制

#### 问题
当前记忆管线只有"存"（FactExtractor）和"取"（SemanticMemory），没有"反思学习"。Generative Agents 证明了定期反思能从碎片记忆中提炼高层洞察，提升行为一致性。

#### 借鉴来源
- Generative Agents：每100步反思 → LLM提出3个高层问题 → 检索相关记忆回答 → 存回记忆
- Letta Sleeptime：memory_rethink 工具在"睡眠"时重组记忆

#### 改造文件
`memory/memory_pipeline.py`

#### 具体改动

```python
# memory/memory_pipeline.py

import asyncio
import json
import logging
from typing import Optional, List, Dict

logger = logging.getLogger(__name__)

class MemoryReflector:
    """
    记忆反思引擎
    
    每20轮对话触发一次，在后台异步执行：
    1. 取最近的重要事实和对话片段
    2. LLM 提炼 2-3 条高层洞察
    3. 存入语义记忆（category=reflection, importance=0.8）
    """
    
    def __init__(self, semantic_memory, episodic_memory, llm_gateway,
                 trigger_rounds: int = 20):
        self.sm = semantic_memory
        self.em = episodic_memory
        self.llm = llm_gateway
        self.trigger_rounds = trigger_rounds
        self._counter = 0
    
    def should_reflect(self) -> bool:
        """检查是否达到反思触发条件"""
        self._counter += 1
        return self._counter >= self.trigger_rounds
    
    async def reflect_async(self, persona_name: str = "十四"):
        """后台执行反思（不阻塞对话）"""
        try:
            # 1. 获取最近的重要记忆
            recent_facts = self._safe_get_facts()
            recent_episodes = self._safe_get_episodes()
            
            if not recent_facts and not recent_episodes:
                self._counter = 0
                return
            
            # 2. 构建反思 prompt
            facts_str = "\n".join(
                f"- {f['fact']}" if isinstance(f, dict) else f"- {f}"
                for f in recent_facts[:20]
            )
            episodes_str = "\n".join(
                f"- {ep.get('metadata', {}).get('summary', str(ep)[:50])}"
                for ep in recent_episodes[:5]
            )
            
            prompt = (
                f"你是{persona_name}。请回顾最近的记忆，反思并提炼洞察。\n\n"
                f"## 关于用户的事实\n{facts_str}\n\n"
                f"## 最近的对话\n{episodes_str}\n\n"
                f"请提炼2-3条高层洞察（每条不超过30字）：\n"
                f"- 用户的性格特点或习惯\n"
                f"- 用户最近的情绪状态\n"
                f"- 你们关系的变化\n\n"
                f"输出JSON格式："
                f'[{{"insight": "...", "category": "personality|emotion|relationship"}}]'
            )
            
            # 3. 调用LLM反思
            result = await self.llm.chat_sync(query=prompt, max_tokens=512)
            insights = json.loads(result)
            
            # 4. 存入语义记忆
            for item in insights:
                if isinstance(item, dict) and "insight" in item:
                    self.sm.add_fact(
                        fact=item["insight"],
                        category=f"reflection_{item.get('category', 'general')}",
                        confidence=0.7,
                        importance=0.8,
                        source="reflection_engine",
                    )
            
            logger.info(f"记忆反思完成: 生成 {len(insights)} 条洞察")
            
        except json.JSONDecodeError:
            logger.warning("反思结果JSON解析失败")
        except Exception as e:
            logger.warning(f"反思执行异常: {e}")
        finally:
            self._counter = 0  # 无论成功失败都重置计数器
    
    def _safe_get_facts(self) -> list:
        try:
            return self.sm.get_facts(limit=30)
        except Exception:
            return []
    
    def _safe_get_episodes(self) -> list:
        try:
            return self.em.search("", top_k=5)
        except Exception:
            return []
```

### 4.3 上下文智能压缩

#### 问题
当前 `ConversationSummarizer` 只有 LLM 摘要模式，没有降级策略。如果 LLM 不可用，上下文就无法压缩。

#### 【核心变更】2级降级取代3级降级
原方案设计了 SLIDING_WINDOW / AGGRESSIVE / TEMPLATE 三级。实际项目中 keep_recent=50 已经够用，AGGRESSIVE 模式没有使用场景。**简化为 LLM + template 两级。**

#### 改造文件

#### 具体改动

```python
# memory/conversation_summarizer.py

class SmartCompressor:
    """
    智能上下文压缩器（2级降级）
    
    降级链：LLM摘要 → template摘要
    - LLM摘要：调用LLM对旧消息做语义压缩
    - template摘要：LLM不可用时，提取关键词+话题
    """
    
    def __init__(self, llm_gateway, keep_recent: int = 20):
        self.llm = llm_gateway
        self.keep_recent = keep_recent
    
    def compress(self, messages: list) -> tuple:
        """
        压缩对话历史
        
        返回: (recent_messages, summary_str)
        """
        if len(messages) <= self.keep_recent:
            return messages, ""  # 不需要压缩
        
        old = messages[:-self.keep_recent]
        recent = messages[-self.keep_recent:]
        
        # 1级: LLM摘要
        summary = self._llm_summarize(old)
        if summary:
            return recent, summary
        
        # 2级(降级): template摘要
        return recent, self._template_summarize(old)
    
    def _llm_summarize(self, messages: list, max_chars: int = 300) -> str:
        """LLM摘要（1级）"""
        try:
            text = "\n".join(
                f"{'用户' if m.get('is_user') else '我'}: {m.get('content', '')}"
                for m in messages[-30:]
            )
            prompt = (
                f"用不超过{max_chars}字总结以下对话要点，"
                f"保留话题、关键信息、情感变化：\n\n{text[-3000:]}"
            )
            result = self.llm.chat_sync(query=prompt, max_tokens=max_chars)
            return result.strip()
        except Exception:
            return None
    
    def _template_summarize(self, messages: list) -> str:
        """template摘要（2级降级）"""
        # 提取话题关键词
        topics = set()
        emotions = set()
        for m in messages:
            content = m.get("content", "")
            if len(content) > 5:
                topics.add(content[:15])
            # 尝试提取情感标签
            emo = m.get("metadata", {}).get("emotion", "")
            if emo:
                emotions.add(emo)
        
        parts = [f"[摘要] 共{len(messages)}条对话"]
        if topics:
            parts.append(f"话题：{'、'.join(list(topics)[:5])}")
        if emotions:
            parts.append(f"情绪变化：{'→'.join(list(emotions)[:3])}")
        
        return " | ".join(parts)
```

#### 验收标准
- [ ] ≤ keep_recent(20) 条消息 → 不压缩
- [ ] > keep_recent 条 → LLM摘要 → 失败则 template 降级
- [ ] 压缩耗时 ≤ 3s（LLM模式）或 ≤ 100ms（template模式）
- [ ] 压缩率记录在日志中
- [ ] 当前用户输入和最近2轮对话始终保留

```python
from collections import deque
from dataclasses import dataclass, field
from typing import Optional

@dataclass
class EmotionHistory:
    """情感历史追踪"""
    recent: deque = field(default_factory=lambda: deque(maxlen=10))
    consecutive: dict = field(default_factory=dict)
    
    def record(self, emotion):
        self.recent.append(emotion)
        # 更新连续计数
        last = self.recent[-2] if len(self.recent) >= 2 else None
        if emotion == last:
            self.consecutive[emotion] = self.consecutive.get(emotion, 0) + 1
        else:
            self.consecutive[emotion] = 1
    
    def consecutive_count(self, emotion) -> int:
        return self.consecutive.get(emotion, 0)
    
    def last(self):
        return self.recent[-1] if self.recent else None

class EmotionPredictor:
    """情感预测器（基于转移先验）"""
    
    # 情感转移先验概率
    TRANSITION_PRIORS = {
        ("NEUTRAL", "HAPPY"): 0.3, ("NEUTRAL", "SAD"): 0.1,
        ("HAPPY", "NEUTRAL"): 0.4, ("HAPPY", "LOVELY"): 0.3,
        ("SAD", "SAD"): 0.3, ("SAD", "TIRED"): 0.2,
        ("ANGRY", "ANGRY"): 0.4, ("ANGRY", "SAD"): 0.2,
        ("LOVELY", "HAPPY"): 0.5, ("TIRED", "NEUTRAL"): 0.4,
    }
    
    def predict_next(self, history: EmotionHistory) -> Optional[str]:
        """预测下一轮情感"""
        if len(history.recent) < 2:
            return None
        
        last = history.last()
        if last is None:
            return None
        
        # 连续3轮同一情感 → 预测持续
        if history.consecutive_count(last) >= 3:
            return str(last)
        
        # 基于先验预测
        candidates = [(next_e, prob) 
                     for (prev, next_e), prob in self.TRANSITION_PRIORS.items()
                     if str(prev) == str(last)]
        
        if candidates:
            return max(candidates, key=lambda x: x[1])[0]
        return None
```

### 5.2 情感转移矩阵补全

#### 问题
当前 `EMOTION_TRANSITION_MATRIX` 只有 8 条转移对（10×10=100 种可能，覆盖率 8%），其余用默认概率 0.5。情感变化不够自然。

#### 【核心变更】规则驱动取代硬编码
原方案要手写 94 条新概率（100 条完整矩阵），容易填错且维护成本高。改用**愉悦度距离规则**：

```
转移概率 = max(0.3, 1.0 - |愉悦度差距| / 2.0)
```

相似情感 → 高转移概率；相反情感 → 低转移概率。**天然覆盖 100% 且行归一化自动满足。**

#### 改造文件
`my_character/emotion_engine.py`

#### 具体改动

```python
# emotion_engine.py ContinuityGuard.check_transition() 改造

# 保留原始8条特例（向后兼容）
EMOTION_TRANSITION_MATRIX = {
    (NEUTRAL, ANGRY): 0.6,
    (NEUTRAL, JEALOUS): 0.5,
    (HAPPY, LOVELY): 0.8,
    (LOVELY, HAPPY): 0.8,
    (LOVELY, JEALOUS): 0.6,
    (CARING, HAPPY): 0.7,
    (TIRED, NEUTRAL): 0.6,
    (SULLEN, NEUTRAL): 0.4,
}

MIN_TRANSITION_PROB = 0.3  # 最低转移概率

# 愉悦度映射（已有）
EMOTION_PLEASURE_MAP = {
    HAPPY: 1.0, LOVELY: 0.9, PLAYFUL: 0.8, CARING: 0.7,
    NEUTRAL: 0.0, TIRED: -0.5, SULLEN: -0.6, SAD: -0.8,
    JEALOUS: -0.9, ANGRY: -1.0,
}

def check_transition(self, old_emotion: Emotion, new_emotion: Emotion) -> bool:
    """
    检查情感转移是否允许
    
    策略（白名单+规则推断）：
    1. 如果在 EMOTION_TRANSITION_MATRIX 中，使用指定概率
    2. 否则，基于愉悦度距离自动推断概率
    """
    key = (old_emotion, new_emotion)
    
    # 白名单：原始8条特例
    if key in EMOTION_TRANSITION_MATRIX:
        prob = EMOTION_TRANSITION_MATRIX[key]
    else:
        # 规则推断：愉悦度差距越小，转移概率越高
        pleasure_old = EMOTION_PLEASURE_MAP.get(old_emotion, 0.0)
        pleasure_new = EMOTION_PLEASURE_MAP.get(new_emotion, 0.0)
        delta = abs(pleasure_old - pleasure_new)
        prob = max(MIN_TRANSITION_PROB, 1.0 - delta / 2.0)
    
    return random.random() < prob
```

#### 设计说明

| 情感差距 | 例子 | 转移概率 | 合理性 |
|----------|------|---------|--------|
| 0.0（相同） | HAPPY→HAPPY | 1.0 | 情感有惯性 |
| ≤ 0.3（相近） | HAPPY→LOVELY | ≥ 0.85 | 开心→撒娇很自然 |
| 0.5-1.0（中等） | NEUTRAL→SAD | 0.5-0.75 | 平常变伤心可能 |
| > 1.5（相反） | HAPPY→ANGRY | ≤ 0.3(MIN) | 开心→生气不自然 |
| 2.0（对立） | HAPPY→SAD | 0.3(MIN) | 极小概率 |

- [ ] 原始8条转移概率值不变（向后兼容）
- [ ] 概率 ∈ [0, 1]
- [ ] 行归一化误差 ≤ 0.01
- [ ] 转移时间 O(1)（查表+简单计算）

```python
# rag_engine.py 新增

def rrf_fusion(vector_results: list, bm25_results: list,
               k: int = 60, top_k: int = 10) -> list:
    """
    RRF（Reciprocal Rank Fusion）融合排序
    
    参数:
        vector_results: [(doc_id, score), ...]
        bm25_results: [(doc_id, score), ...]
        k: RRF 常数（默认60）
        top_k: 返回结果数
    
    返回:
        [(doc_id, rrf_score), ...] 按分数降序
    """
    vector_ranks = {doc: rank for rank, (doc, _) in enumerate(vector_results)}
    bm25_ranks = {doc: rank for rank, (doc, _) in enumerate(bm25_results)}
    
    all_docs = set(vector_ranks.keys()) | set(bm25_ranks.keys())
    
    rrf_scores = {}
    for doc in all_docs:
        v_rank = vector_ranks.get(doc, len(vector_results))
        b_rank = bm25_ranks.get(doc, len(bm25_results))
        rrf_scores[doc] = 1.0 / (k + v_rank + 1) + 1.0 / (k + b_rank + 1)
    
    sorted_docs = sorted(rrf_scores.items(), key=lambda x: x[1], reverse=True)
    return sorted_docs[:top_k]
```

**步骤3：替换现有的简单加权融合**

在检索方法中将：
```python

# score = vector_score * 0.7 + keyword_score * 0.3
```
替换为：
```python

```

### 6.1 记忆情感标注检索

#### 问题
记忆存入时已有 `emotion_tag`（情感标注），但检索时没有利用它。用户表达开心时，优先找到开心时发生的记忆，回复会更贴切。

#### 具体改动

改造 `retrieve_context()` 方法，在语义检索结果上增加情感加权：

```python
# memory/memory_pipeline.py MemoryPipeline

async def retrieve_context(self, query: str, emotion_state=None, top_k=5):
    """检索上下文（带情感加权）"""
    
    # 1. 基础检索（不变）
    episodic_results = await self.episodic_mem.search(query, top_k=top_k * 2)
    semantic_results = await self.semantic_mem.search(query, top_k=top_k * 2)
    
    # 2. 情感加权（新增）
    if emotion_state and hasattr(emotion_state, 'primary_emotion'):
        current_emo = self._get_emotion_name(emotion_state.primary_emotion)
        
        episodic_results = [
            self._apply_emotion_weight(r, current_emo)
            for r in episodic_results
        ]
        semantic_results = [
            self._apply_emotion_weight(r, current_emo)
            for r in semantic_results
        ]
        
        # 重排序
        episodic_results.sort(key=lambda x: x['score'], reverse=True)
        semantic_results.sort(key=lambda x: x['score'], reverse=True)
    
    return episodic_results[:top_k], semantic_results[:top_k]

def _get_emotion_name(self, emotion) -> str:
    """获取情感名称字符串"""
    if hasattr(emotion, 'value'):
        return emotion.value
    return str(emotion)

def _apply_emotion_weight(self, result: dict, current_emotion: str) -> dict:
    """
    对检索结果应用情感加权
    
    权重策略：
    - 相同情感: score * 1.3（+30%）
    - 相关情感: score * 1.1（+10%）
    - 无关情感: score * 1.0（不变）
    """
    stored_emotion = result.get('metadata', {}).get('emotion_tag', '')
    if not stored_emotion:
        return result  # 无情感标注的历史数据，权重不变
    
    if stored_emotion == current_emotion:
        result['score'] = result['score'] * 1.3
    elif self._is_related_emotion(stored_emotion, current_emotion):
        result['score'] = result['score'] * 1.1
    
    return result

def _is_related_emotion(self, e1: str, e2: str) -> bool:
    """判断两种情感是否相关"""
    groups = [
        {"开心", "调皮", "撒娇", "温柔"},
        {"伤心", "疲惫", "平常"},
        {"生气", "吃醋", "傲娇"},
    ]
    for group in groups:
        if e1 in group and e2 in group:
            return True
    return False
```

#### 验收标准
- [ ] 记忆存入时自动标注当前情感（如无则跳过）
- [ ] 检索时相同情感 +30% 权重，相关情感 +10%
- [ ] 情感加权不导致完全过滤不匹配记忆（只是排序靠后）
- [ ] 情感引擎不可用时退化为纯语义检索
