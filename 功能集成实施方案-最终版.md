```python
def send(self, reply: Reply, context: Context):
    """发送消息 - 添加表情包支持"""
    receiver = context.get("receiver", "")
    msg = context.get("msg")
    context_token = self._get_context_token(receiver, msg)

    if not context_token:
        logger.error(f"[Weixin] No context_token for receiver={receiver}")
        return

    # 新增: 表情包发送
    if reply.type == ReplyType.STICKER:
        self._send_sticker(reply.content, receiver, context_token)
    elif reply.type == ReplyType.TEXT:
        # 检查是否附带表情包
        if hasattr(reply, 'sticker') and reply.sticker:
            self._send_text_with_sticker(reply.content, reply.sticker, receiver, context_token)
        else:
            self._send_text(reply.content, receiver, context_token)
    elif reply.type in (ReplyType.IMAGE_URL, ReplyType.IMAGE):
        self._send_image(reply.content, receiver, context_token)
    elif reply.type == ReplyType.FILE:
        self._send_file(reply.content, receiver, context_token)
    elif reply.type in (ReplyType.VIDEO, ReplyType.VIDEO_URL):
        self._send_video(reply.content, receiver, context_token)
    else:
        logger.warning(f"[Weixin] Unsupported reply type: {reply.type}, fallback to text")
        self._send_text(str(reply.content), receiver, context_token)


def _send_sticker(self, sticker_path: str, receiver: str, context_token: str):
    """
    发送表情包
    
    策略:
    - 先用send_image_item尝试发送（测试GIF动画效果）
    - 如果无效，后续考虑新增send_gif_item方法
    
    ⚠️ 需要实际测试验证GIF动画效果
    """
    import os
    
    if not sticker_path or not os.path.exists(sticker_path):
        logger.warning(f"[Weixin] Sticker file not found: {sticker_path}")
        return
    
    ext = os.path.splitext(sticker_path)[1].lower()
    
    # 统一使用图片通道发送
    # GIF会通过send_image_item发送，需要测试是否显示动画
    self._send_image(sticker_path, receiver, context_token)
    
    if ext == '.gif':
        logger.info(f"[Weixin] GIF sticker sent via image channel to {receiver}")
    else:
        logger.info(f"[Weixin] Static sticker ({ext}) sent to {receiver}")


def _send_text_with_sticker(self, text: str, sticker: dict, receiver: str, context_token: str):
    """
    文本+表情包组合发送
    
    顺序: 先发文本，再发表情包(间隔0.3s)
    """
    # 1. 发送文本
    self._send_text(text, receiver, context_token)
    
    # 2. 短暂延迟后发送表情包
    import time
    time.sleep(0.3)
    
    # 3. 发送表情包
    sticker_path = sticker.get('path', '')
    if sticker_path:
        self._send_sticker(sticker_path, receiver, context_token)
```

```python
"""情感驱动语音合成 - 参数映射"""

from __future__ import annotations

import logging
from typing import Any

logger = logging.getLogger("shisi.voice_ext.emotion_tts")


class EmotionVoiceMapper:
    """
    情感→语音参数映射器
    
    ✅ Edge-TTS支持: rate（语速）, volume（音量）
    ❌ Edge-TTS不支持: pitch（音调）- 已移除
    """
    
    # 情感参数映射表（基于实际验证）
    _EMOTION_PARAMS = {
        "开心": {
            "rate": "+10%",       # 稍快
            "volume": "+10%",     # 稍大声
        },
        "撒娇": {
            "rate": "-5%",        # 稍慢
            "volume": "+5%",      # 稍大
        },
        "温柔": {
            "rate": "-10%",       # 慢
            "volume": "-5%",      # 小
        },
        "伤心": {
            "rate": "-15%",       # 较慢
            "volume": "-10%",     # 较小声
        },
        "生气": {
            "rate": "+15%",       # 快
            "volume": "+15%",     # 大声
        },
        "害怕": {
            "rate": "+10%",       # 稍快
            "volume": "-5%",      # 稍小
        },
        "害羞": {
            "rate": "-8%",        # 稍慢
            "volume": "-10%",     # 小
        },
        "傲娇": {
            "rate": "+0%",
            "volume": "+0%",
        },
        "平常": {
            "rate": "+0%",
            "volume": "+0%",
        },
    }
    
    def __init__(self):
        self._character_overrides: dict[str, dict[str, Any]] = {}
    
    def get_params(self, emotion: str, character_id: str = "") -> dict[str, Any]:
        """获取情感对应的语音参数"""
        params = self._EMOTION_PARAMS.get(emotion, self._EMOTION_PARAMS["平常"]).copy()
        
        if character_id and character_id in self._character_overrides:
            override = self._character_overrides[character_id].get(emotion, {})
            params.update(override)
        
        return params
    
    def apply_to_edge_tts(self, emotion: str, character_id: str = "") -> dict[str, str]:
        """
        生成Edge-TTS参数
        
        Returns:
            {"rate": "+10%", "volume": "+5%"}
        """
        params = self.get_params(emotion, character_id)
        return {
            "rate": params.get("rate", "+0%"),
            "volume": params.get("volume", "+0%"),
            # ❌ 不返回pitch（不支持）
        }
```

```python
async def synthesize(self, text: str, emotion: str = "", **kwargs) -> Optional[bytes]:
    """
    合成语音 - 支持情感参数
    
    Args:
        text: 文本
        emotion: 情感状态(如"开心"、"伤心")
        **kwargs: 其他参数
    """
    if not self._enabled or not text:
        return None
    
    # 情感参数注入（仅Edge-TTS）
    if emotion and self._current_engine == "edge-tts":
        from shisi.voice_ext.emotion_tts import EmotionVoiceMapper
        mapper = EmotionVoiceMapper()
        emotion_params = mapper.apply_to_edge_tts(emotion)
        kwargs.update(emotion_params)
        logger.debug(f"[TTS] 情感参数注入: {emotion} -> {emotion_params}")
    
    # 原有逻辑
    if self._current_engine and self._current_engine in self._providers:
        provider = self._providers[self._current_engine]
        result = await provider.synthesize(text, **kwargs)
        if result is not None:
            self._synthesize_count += 1
            return result
    
    # 降级逻辑
    for name, provider in self._providers.items():
        if name == self._current_engine:
            continue
        result = await provider.synthesize(text, **kwargs)
        if result is not None:
            logger.warning(f"[TTS] 降级到 {name}")
            self._current_engine = name
            self._synthesize_count += 1
            return result
    
    return None
```

### 5.2 实现方案（重大调整）

#### Step 1: 音色训练管理器（配置文件方式）

**文件**: `voice/voice_training.py`

```python
"""
音色训练管理器 — 用户手动上传音频 → GPT-SoVITS训练

流程:
1. 接收用户上传的音频文件
2. 格式统一转换为WAV(16kHz单声道)
3. 可选降噪
4. 自动切片(3~10秒)
5. ASR自动标注(Whisper/FunASR)
6. 生成GPT-SoVITS配置文件(YAML)
7. 调用GPT-SoVITS训练脚本
8. 训练完成自动绑定角色

⚠️ 重大调整: GPT-SoVITS使用配置文件而非命令行参数
"""

import yaml
import json

class VoiceTrainingManager:
    """音色训练管理器 - 配置文件方式"""
    
    async def train(
        self,
        model_name: str,
        gpt_sovits_dir: str = "third_party/GPT-SoVITS",
        epochs: int = 8,
        batch_size: int = 4,
    ) -> dict[str, Any]:
        """
        调用GPT-SoVITS训练（配置文件方式）
        
        ❌ 原方案: 调用HTTP API（不存在）
        ❌ 原方案: 命令行参数（不正确）
        ✅ 修正: 生成YAML配置文件 + 调用训练脚本
        """
        safe_name = _SAFE_NAME_RE.sub("_", model_name)
        
        # 1. 准备数据文件
        list_file = self._models / f"{safe_name}.list"
        if not list_file.exists():
            return {"error": "训练数据集不存在，请先完成预处理和标注"}
        
        # 2. 生成语义和音素文件（GPT-SoVITS格式）
        semantic_path, phoneme_path = self._prepare_training_data(list_file, safe_name)
        
        # 3. 生成GPT训练配置文件 (s1longer.yaml)
        s1_config = self._generate_s1_config(
            model_name=safe_name,
            semantic_path=semantic_path,
            phoneme_path=phoneme_path,
            epochs=epochs,
        )
        s1_config_path = self._models / f"{safe_name}_s1.yaml"
        with open(s1_config_path, 'w', encoding='utf-8') as f:
            yaml.dump(s1_config, f, default_flow_style=False)
        
        # 4. 生成SoVITS训练配置文件 (s2.json)
        s2_config = self._generate_s2_config(
            model_name=safe_name,
            list_file=list_file,
            epochs=epochs,
            batch_size=batch_size,
        )
        s2_config_path = self._models / f"{safe_name}_s2.json"
        with open(s2_config_path, 'w', encoding='utf-8') as f:
            json.dump(s2_config, f, indent=2)
        
        self._state.update({
            "status": "training",
            "step": "GPT-SoVITS训练中（配置文件方式）",
            "progress": 0.7,
        })
        
        try:
            gpt_path = Path(gpt_sovits_dir)
            
            # Stage 1: GPT训练
            s1_script = gpt_path / "GPT_SoVITS" / "s1_train.py"
            if not s1_script.exists():
                return {"error": f"GPT训练脚本不存在: {s1_script}"}
            
            cmd1 = ["python", str(s1_script), "-c", str(s1_config_path)]
            
            proc1 = await asyncio.create_subprocess_exec(
                *cmd1,
                stdout=asyncio.subprocess.PIPE,
                stderr=asyncio.subprocess.PIPE,
            )
            stdout1, stderr1 = await proc1.communicate()
            
            if proc1.returncode != 0:
                error_msg = stderr1.decode()[:500] if stderr1 else "GPT训练失败"
                self._state.update({"status": "error", "error": error_msg})
                return {"status": "error", "error": error_msg}
            
            # Stage 2: SoVITS训练
            s2_script = gpt_path / "GPT_SoVITS" / "s2_train.py"
            cmd2 = ["python", str(s2_script), "-c", str(s2_config_path)]
            
            proc2 = await asyncio.create_subprocess_exec(
                *cmd2,
                stdout=asyncio.subprocess.PIPE,
                stderr=asyncio.subprocess.PIPE,
            )
            stdout2, stderr2 = await proc2.communicate()
            
            if proc2.returncode != 0:
                error_msg = stderr2.decode()[:500] if stderr2 else "SoVITS训练失败"
                self._state.update({"status": "error", "error": error_msg})
                return {"status": "error", "error": error_msg}
            
            # 查找输出模型
            output_dir = self._models / safe_name
            pth_files = list(output_dir.glob("**/*.pth"))
            
            if pth_files:
                model_path = str(max(pth_files, key=lambda p: p.stat().st_mtime))
                self._state.update({
                    "status": "done",
                    "step": "训练完成",
                    "progress": 1.0,
                    "result": {"model_path": model_path},
                })
                return {"status": "success", "model_path": model_path}
            else:
                self._state.update({"status": "error", "error": "未找到输出模型"})
                return {"status": "error", "error": "未找到输出模型"}
                
        except Exception as e:
            logger.exception("[VoiceTraining] 训练失败: %s", e)
            self._state.update({"status": "error", "error": str(e)})
            return {"status": "error", "error": str(e)}
    
    def _generate_s1_config(
        self,
        model_name: str,
        semantic_path: Path,
        phoneme_path: Path,
        epochs: int,
    ) -> dict:
        """生成GPT训练配置文件 (s1longer.yaml)"""
        return {
            "train": {
                "exp_name": model_name,
                "epochs": epochs,
                "if_save_latest": True,
                "if_save_every_weights": True,
                "half_weights_save_dir": str(self._models / model_name / "weights"),
                "output_dir": str(self._models / model_name / "logs"),
                "seed": 42,
                "precision": "16-mixed",
                "save_every_n_epoch": 1,
            },
            "train_semantic_path": str(semantic_path),
            "train_phoneme_path": str(phoneme_path),
        }
    
    def _generate_s2_config(
        self,
        model_name: str,
        list_file: Path,
        epochs: int,
        batch_size: int,
    ) -> dict:
        """生成SoVITS训练配置文件 (s2.json)"""
        return {
            "train": {
                "exp_name": model_name,
                "epochs": epochs,
                "batch_size": batch_size,
                "if_save_latest": True,
                "if_save_every_weights": True,
                "half_weights_save_dir": str(self._models / model_name / "weights"),
                "output_dir": str(self._models / model_name / "logs"),
            },
            "list_file": str(list_file),
        }
```
